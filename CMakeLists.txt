cmake_minimum_required(VERSION 3.24)
project(WeChatAXRIntegration LANGUAGES CXX)
include(CTest)
add_subdirectory(AXR)
if(APPLE)
  # Application-specific, opt-in owner tool; never part of the AXR SDK export.
  add_library(wechat-key-observer SHARED adapters/macos/KeyObserver.cpp)
  target_compile_features(wechat-key-observer PRIVATE cxx_std_20)
  if(BUILD_TESTING)
    add_executable(wechat-key-fixture adapters/macos/KeyObserverFixture.cpp)
    target_compile_features(wechat-key-fixture PRIVATE cxx_std_20)
  endif()
endif()
if(WIN32)
  add_executable(wechat-uia adapters/windows/WeChatUIA.cpp)
  target_compile_features(wechat-uia PRIVATE cxx_std_20)
  target_link_libraries(wechat-uia PRIVATE Stellaria::AXR ole32 oleaut32 uiautomationcore user32)
endif()
